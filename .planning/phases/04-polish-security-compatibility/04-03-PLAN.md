   ```python
   """TOTP-based message encryption/decryption (stub implementation).

   All channels use the current 6-digit TOTP as a symmetric key. Production should
   upgrade to AES-GCM or ChaCha20-Poly1305 with proper key derivation. This stub
   uses simple XOR for reversibility during development and testing.
   """
   import logging
   logger = logging.getLogger(__name__)

   def _totp_to_key(totp: str) -> bytes:
       """Convert a 6-digit TOTP string into a 16-byte key."""
       if not totp or not totp.isdigit() or len(totp) != 6:
           raise ValueError("Invalid TOTP code")
       numeric = int(totp)
       # Convert to 8-byte big-endian, then repeat to 16 bytes
       base = numeric.to_bytes(8, 'big')
       key = (base * 2)[:16]
       return key

   def encrypt_message(message: bytes, totp: str) -> bytes:
       """Encrypt a message using the TOTP-derived key.

       Args:
           message: Plaintext bytes.
           totp: 6-digit TOTP code string.

       Returns:
           Ciphertext bytes same length as message.

       Raises:
           ValueError: If TOTP code is invalid.
       """
       key = _totp_to_key(totp)
       # Repeat key to message length
       keystream = (key * ((len(message) // len(key)) + 1))[:len(message)]
       return bytes(a ^ b for a, b in zip(message, keystream))

   def decrypt_message(ciphertext: bytes, totp: str) -> bytes:
       """Decrypt a message using the TOTP-derived key (XOR is symmetric)."""
       return encrypt_message(ciphertext, totp)
   ```
3. Do not add any external dependencies; use only stdlib.
4. Ensure the functions raise `ValueError` on invalid TOTP.
